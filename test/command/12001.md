With `rebase_relative_paths`, relative paths in a footnote definition
or grid table that ends a file must be rebased against that file's
directory, not the next input file's (#12001).

```
% pandoc -f markdown+rebase_relative_paths -t native command/12001/note/book.md command/12001/addendum/addendum.md
^D
[ Para
    [ Str "Text"
    , Note
        [ Para
            [ Image
                ( "" , [] , [] )
                [ Str "C" ]
                ( "command/12001/note/img/image.png" , "" )
            , Space
            , Link
                ( "" , [] , [] )
                [ Str "doc" ]
                ( "command/12001/note/files/doc.pdf" , "" )
            ]
        ]
    ]
, Header 1 ( "addendum" , [] , [] ) [ Str "Addendum" ]
]
```

    [ ( AlignDefault , ColWidth 0.3055555555555556 ) ]
    (TableHead ( "" , [] , [] ) [])

        ( "" , [] , [] )

            ( "" , [] , [] )

                ( "" , [] , [] )
                AlignDefault

                (ColSpan 1)

                    ( "" , [] , [] )
                    (Caption Nothing [ Plain [ Str "C" ] ])
                    [ Plain
                        [ Image

                            [ Str "C" ]
                            ( "command/12001/grid/img/image.png"
                            , ""
                            )
                        ]
                    ]
                ]
            ]
        ]
    ]
    (TableFoot ( "" , [] , [] ) [])
, Header 1 ( "addendum" , [] , [] ) [ Str "Addendum" ]
]
